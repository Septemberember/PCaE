public class AddLoop_Mutant3 {
    // Utility: compute sum using loop-based accumulation
    // The method demonstrates two branches dependent on y
    //@ requires Integer.MIN_VALUE <= x + 2*y && x + 2*y <= Integer.MAX_VALUE && y != Integer.MIN_VALUE && y >= 0;
    //@ requires Integer.MIN_VALUE <= x + y && x + y <= Integer.MAX_VALUE && y != Integer.MIN_VALUE && y <= 0;
    //@ ensures y > 0 ==> \result == x + 2*y;
    //@ ensures y <= 0 ==> \result == x + y;
    public static int AddLoop(int x, int y) {
        int sum = x;
        // Start with the provided x as the base and update in the loop
        // For y > 0, each iteration increases sum by 2
        if (y > 0) {
            int n = y;
            // Iterate n times, incrementing sum by 2 per iteration
            //@ maintaining sum == x + 2*y - 2*n && 0 <= n;
            //@ decreases n;
            while (n > 0) {
                sum = sum + 2;
                n = n - 1;
            }
        } else {
            int n = -y;
            // Iterate n times, decrementing sum by 1 per iteration
            //@ maintaining sum == x + y + n && 0 <= n;
            //@ decreases n;
            while (n > 0) {
                sum = sum - 1;
                n = n - 1;
            }
        }
        return sum;
    }
}
