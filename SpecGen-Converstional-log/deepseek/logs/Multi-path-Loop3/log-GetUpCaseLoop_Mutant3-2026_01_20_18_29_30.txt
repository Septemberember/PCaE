    //@ ensures \result != 0 ==> (\result == c - 'a' + 'A');
    public static char getUpCase(char c) {

        if(c < 'f' || c > 'z') {
            return 0;
        }
        // Start with the uppercase offset from 'A'
        char t = 'F';

        //@ maintaining t >= 'F' && t <= 'Z' + 1;
        //@ maintaining t <= 'Z' ==> t != c - 'a' + 'A';
        //@ decreases 'Z' - t + 1;
        while(t <= 'Z' && t != c - 'a' + 'A') {
