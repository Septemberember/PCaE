public class PowerOfTwo_Original {
    //@ ensures \result <==> (n > 0 && (\exists int k; 0 <= k && k <= 30; n == (1 << k)));
    //@ ensures \result <==> (n > 0 && (n & (n - 1)) == 0);
    public static boolean isPowerOfTwo(int n) {
        return n > 0 && (n & (n - 1)) == 0;
    }
}
