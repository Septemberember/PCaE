public class AbsSeq_Mutant1 {
    //@ requires num != Integer.MIN_VALUE;
    //@ ensures \result == (num < 0 ? -num : num);
    //@ ensures \result >= 0;
    public static int Abs(int num) {
        if (num <= 0) {
            num = -num;
	    return num;
        } else {
            return num;
        }
    }
}
