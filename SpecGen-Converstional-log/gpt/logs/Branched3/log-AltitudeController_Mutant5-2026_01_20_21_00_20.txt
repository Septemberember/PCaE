    /**
     * Altitude controller logic for discrete gain adjustments.

    // Altitude controller: uses simple, discrete control actions based on error magnitude.
    // This mutant uses coarse positive or negative gains for altitude adjustments.
    // Compute control signal based on altitude error.
    /*@ public normal_behavior
      @   // Prevent integer overflow in: error = targetHeight - currentHeight, and in: -error
      @   requires ( (long)targetHeight - (long)currentHeight ) <= Integer.MAX_VALUE;
      @   requires ( (long)targetHeight - (long)currentHeight ) >= Integer.MIN_VALUE + 1;
      @   assignable \nothing;
      @   ensures (\let int error = targetHeight - currentHeight;
      @           \let int absError = (error < 0 ? -error : error);
      @             (absError > 30 ==> \result == (error > 0 ? 5 : -4)) &&
      @             (absError <= 30 && absError > 20 ==> \result == (error > 0 ? 3 : -3)) &&
      @             (absError <= 20 && absError > 10 ==> \result == (error > 0 ? 2 : -2)) &&
      @             (absError <= 10 ==> \result == error)
      @          );
      @*/
    public static int altitudeController(int currentHeight,int targetHeight) {
        int error = targetHeight - currentHeight;
        int absError = 0;
        absError = error < 0 ? -error : error;
        int controlSignal = 0;
        if (absError > 30) {
            controlSignal = error > 0 ? 5 : -4; //Changed -5 to -4
        } else if (absError > 20) {
            controlSignal = error > 0 ? 3 : -3;
        } else if (absError > 10) {
            controlSignal = error > 0 ? 2 : -2;
        } else {
            controlSignal = error;
        }
        return controlSignal;
    }
}
