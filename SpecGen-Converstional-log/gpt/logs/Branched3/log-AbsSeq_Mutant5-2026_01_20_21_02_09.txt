public class AbsSeq_Mutant5 {

    // Class containing a straightforward absolute value utility.

    //@ ensures (\old(num) < 0 ==> \result == -\old(num)) && (\old(num) >= 0 ==> \result == \old(num));
    //@ ensures \result >= 0;
    //@ ensures \result == (\old(num) < 0 ? -\old(num) : \old(num));
    public static int Abs(int num) {
        // If negative, convert to positive
        if (num < 0) {
            num = -num;
        } 

        return num;
    }
}
