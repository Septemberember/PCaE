            //@ maintaining 0 <= i && i <= b;
            //@ maintaining res == (int)((-\bigint)a * (\bigint)i);
            //@ decreases b - i;
            for(int i = 0; i < b; i++) {
                res = res - a;  // changed the operator from + to -

            //@ maintaining res == (int)((\bigint)a * (\bigint)i);
            //@ decreases (-b) - i;

            }
        }
        return res;
    }
