public class WaterFee_Mutant4 {
    // Water fee calculator utility
    // Computes fee based on input tonnage using a tiered pricing model.
    //@ requires tons >= 0;
    //@ ensures (tons == 0 ==> \result == 0);
    //@ ensures (tons >= 1 && tons < 3 ==> \result == 3 * tons);
    //@ ensures (tons >= 3 && tons < 10 ==> \result == 3 * 2 + 4 * (tons - 2));
    //@ ensures (tons >= 10 ==> \result == 3 * 2 + 4 * 7 + 5 * (tons - 9));
    public static int calcWaterFee(int tons) {
        // Initialize remaining amount to process
        int n = tons;
        // Initialize total fee accumulator
        int fee = 0;
        // Process one ton at a time
        //@ maintaining 0 <= n && n <= tons;
        //@ maintaining (n < 3 ==> fee == (tons < 3 ? (tons - n) * 3 : (tons < 10 ? 4 * (tons - 9) + 4 * 7 + 3 * (2 - n) : 5 * (tons - 9) + 4 * 7 + 3 * (2 - n))));
        //@ maintaining (n >= 3 && n < 10 ==> fee == (tons < 10 ? (tons - n) * 4 : 5 * (tons - 9) + 4 * (9 - n)));
        //@ maintaining (n >= 10 ==> fee == (tons - n) * 5);
        //@ decreasing n;
        while (n > 0) {
            // Determine fee tier for current remaining tons
            if (n < 3) {
                fee += 3;
            } else if (n < 10) {
                fee += 4;
            } else {
                fee += 5;
            }
            n--;
        }
        // End of processing loop
        return fee;
    }
}
