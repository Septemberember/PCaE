public class WaterFee_Mutant3 {
    // This class provides a simple fee calculation based on water consumption.
    //@ requires tons >= 0;
    //@ ensures (tons <= 3) ==> \result == tons * 1;
    //@ ensures (tons > 3 && tons <= 10) ==> \result == 3 * 1 + (tons - 3) * 4;
    //@ ensures (tons > 10) ==> \result == 3 * 1 + 7 * 4 + (tons - 10) * 5;
    public static int calcWaterFee(int tons) {
        // Use a local copy to avoid modifying the input parameter directly
        int n = tons;

        // Track the accumulated fee
        int fee = 0;

        //@ maintaining 0 <= n && n <= tons;
        //@ maintaining (tons > 10 && n >= 10) ==> fee == (tons - n) * 5;
        //@ maintaining (tons > 10 && n < 10 && n >= 3) ==> fee == (tons - 10) * 5 + (10 - n) * 4;
        //@ maintaining (tons > 10 && n < 3) ==> fee == (tons - 10) * 5 + 7 * 4 + (3 - n) * 1;
        //@ maintaining (tons <= 10 && tons > 3 && n >= 3) ==> fee == (tons - n) * 4;
        //@ maintaining (tons <= 10 && tons > 3 && n < 3) ==> fee == (tons - 3) * 4 + (3 - n) * 1;
        //@ maintaining (tons <= 3) ==> fee == (tons - n) * 1;
        //@ decreases n;
        while (n > 0) {
            if (n > 10) {
                fee += 5;
            } else if (n > 3) {
                fee += 4;
            } else {
                fee += 1;
            }
            // Move to the next ton
            n--;
        }

        // Return the computed total fee
        return fee;
    }
}
