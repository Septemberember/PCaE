/**** Simple parity utility for DivisorGame_Mutant4. */
public class DivisorGame_Mutant4 {
    // Entry point: computes parity of n
    //@ ensures \result == 0 <==> n % 2 == 0;
    //@ ensures \result == 1 || \result == -1 <==> n % 2 != 0;
