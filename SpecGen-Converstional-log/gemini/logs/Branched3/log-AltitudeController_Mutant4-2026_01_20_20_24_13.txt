public class AltitudeController_Mutant4 {
    //@ requires (long)targetHeight - currentHeight <= Integer.MAX_VALUE;
    //@ requires (long)targetHeight - currentHeight >= Integer.MIN_VALUE;
    //@ requires targetHeight - currentHeight != Integer.MIN_VALUE;
    //@ ensures (targetHeight - currentHeight >= 30) ==> \result == 5;
    //@ ensures (targetHeight - currentHeight <= -30) ==> \result == -5;
    //@ ensures (targetHeight - currentHeight < 30 && targetHeight - currentHeight > 20) ==> \result == 3;
    //@ ensures (targetHeight - currentHeight > -30 && targetHeight - currentHeight < -20) ==> \result == -3;
    //@ ensures (targetHeight - currentHeight <= 20 && targetHeight - currentHeight > 10) ==> \result == 2;
    //@ ensures (targetHeight - currentHeight >= -20 && targetHeight - currentHeight < -10) ==> \result == -2;
    //@ ensures (targetHeight - currentHeight <= 10 && targetHeight - currentHeight >= -10) ==> \result == (targetHeight - currentHeight);
    public static int altitudeController(int currentHeight,int targetHeight) {
        int error = targetHeight - currentHeight;
        int absError = 0;
        absError = error < 0 ? -error : error;
        int controlSignal = 0;
        if (absError >= 30) { //Changed > to >=
            controlSignal = error > 0 ? 5 : -5;
        } else if (absError > 20) {
            controlSignal = error > 0 ? 3 : -3;
        } else if (absError > 10) {
            controlSignal = error > 0 ? 2 : -2;
        } else {
            controlSignal = error;
        }
        return controlSignal;
    }
}
