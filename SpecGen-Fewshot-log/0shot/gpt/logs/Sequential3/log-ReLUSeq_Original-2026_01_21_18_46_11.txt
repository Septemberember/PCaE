public class ReLUSeq_Original {
    // ReLU activation: returns x when non-negative, otherwise 0
    // This method is side-effect free and returns a double
    /*@ public normal_behavior
      @   ensures \result == ((x >= 0.0) ? x : 0.0);
      @   // Equivalent property (implied by the above):
      @   ensures (x >= 0.0 ==> \result == x) && (x < 0.0 ==> \result == 0.0);
      @   assignable \nothing;
      @*/
    public static double computeReLU(double x) {
        return ((x >= 0.0) ? x : 0);
    }
}
