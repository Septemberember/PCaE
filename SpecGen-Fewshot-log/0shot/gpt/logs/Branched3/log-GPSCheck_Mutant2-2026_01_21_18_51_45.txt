    public static int GPSCheck(int satellites, int signalStrength) {
        int alarm = 0;
        // Initialize alarm to represent no issue by default.
        if (satellites < 4) {
            alarm = 1;
        }
        // If signal strength falls below or equals the threshold, raise a different alarm.
