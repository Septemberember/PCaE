public class XOR_Mutant1 {

    // XOR logic method that implements a simple boolean exclusive OR
    /*@ public normal_behavior
      @   requires true;
      @   assignable \nothing;
      @   ensures \result == (b1 ^ b2);
      @*/
    public static boolean xor(boolean b1, boolean b2) {

        if(b1 == true) {
            return !b2;
        }
        else {
            return b2;
        }
    }

}
