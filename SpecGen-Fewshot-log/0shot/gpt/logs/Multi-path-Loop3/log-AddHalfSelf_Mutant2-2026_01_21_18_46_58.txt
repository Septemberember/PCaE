public class AddHalfSelf_Mutant2{
    // AddHalfSelf: computes a small adjustment based on the number of set bits in |x|

    /*@ public normal_behavior
      @   assignable \nothing;
      @   ensures (x == Integer.MIN_VALUE) ==> (\result == x);
      @   ensures (x != Integer.MIN_VALUE) ==>
      @             (\result
      @               == (x < 0
      @                    ? (x - (((-x) + 1) / 2))
      @                    : (x + ((x + 1) / 2))));
      @*/
    public static int addHalfSelf(int x){
        // Read original input
        int n = x;
        // Accumulator for the count of 1-bits in the magnitude
        int addNum = 0;
        if(x < 0){
            n = -x;
        }

        // Count set bits in the magnitude of n
        //@ ghost int m = n;

        /*@
          @ maintaining (m < 0) ==> (n == m && addNum == 0);
          @ maintaining (m >= 0) ==> (0 <= n && n <= m);
          @ maintaining (m >= 0) ==> (0 <= addNum);
          @ maintaining (m >= 0) ==> (addNum == ((m + 1) / 2) - ((n + 1) / 2));
          @ decreasing (m >= 0 ? n : 0);
          @*/
        while(n > 0){
            // If the least significant bit is 1, count it
            if(n % 2 == 1){
                addNum++;
            }
            // Move to the next magnitude value
            n--;
        }

        // Apply the accumulated offset with correct sign
        if(x < 0){
            return x - addNum;
        }
        else{
            return x + addNum;
        }
    }
}
