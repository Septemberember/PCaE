    public static double estimateSpeed(int distance,int time) {
        int adjusted = distance;
        double speed = 0.0;
        boolean valid = true;
        int bias = 2;

        adjusted += 10;
        adjusted *= 2;
        adjusted -= bias;
        adjusted = adjusted + 5;

                if (distance > 20) {
                    adjusted = adjusted + distance - 4;
                    speed = adjusted * 0.75;
                    adjusted -= 5;

                    if (time > 0) {
                        adjusted += 2;

                            adjusted += 4;
