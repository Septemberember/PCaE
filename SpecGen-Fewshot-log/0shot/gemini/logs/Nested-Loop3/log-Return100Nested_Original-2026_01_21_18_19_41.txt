/*
 * Utility: demonstrate a simple nested-loop computation.
 * This class contains a method that returns the constant value 100
 * by totaling 100 iterations across two nested loops.
 */

public class Return100Nested_Original {
    /*@ public normal_behavior
      @   ensures \result == 100;
      @   assignable \nothing;
      @*/
    public static int return100 () {
        int res = 0;
        /*@ maintaining 0 <= i && i <= 10;
          @ maintaining res == i * 10;
          @ decreases 10 - i;
          @*/
        for(int i = 0; i < 10; i++) {
            /*@ maintaining 0 <= j && j <= 10;
              @ maintaining res == (i * 10) + j;
              @ decreases 10 - j;
              @*/
            for(int j = 0; j < 10; j++) {
                res = res + 1;
            }
        }
        return res;
    }
}
