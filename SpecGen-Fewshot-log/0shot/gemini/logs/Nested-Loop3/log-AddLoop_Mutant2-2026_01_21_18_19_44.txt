public class AddLoop_Mutant2 {
    // Branch overview
    // - if y > 0, increment sum by 1, for y iterations
    // - else, decrement sum by 2, for -y iterations
    /*@ public normal_behavior
      @   requires (y > 0 ==> (long)x + y <= Integer.MAX_VALUE) && 
      @            (y <= 0 ==> (long)x + 2L * y >= Integer.MIN_VALUE);
      @   ensures (y > 0 ==> \result == x + y) && (y <= 0 ==> \result == x + 2 * y);
      @   assignable \nothing;
      @*/
    public static int AddLoop(int x, int y) {
        // Initialize accumulator with x
        int sum = x;
        if (y > 0) {
            int n = y;
            /*@ maintaining 0 <= n && n <= y;
              @ maintaining sum == x + (y - n);
              @ decreasing n;
              @*/
            while (n > 0) {
                sum = sum + 1;
                n = n - 1;
            }
        } else {
            int n = -y;
            /*@ maintaining 0 <= n && n <= -y;
              @ maintaining sum == x - 2 * (-y - n);
              @ decreasing n;
              @*/
            while (n > 0) {
                sum = sum - 2;
                n = n - 1;
            }
        }
        return sum;
    }
}
