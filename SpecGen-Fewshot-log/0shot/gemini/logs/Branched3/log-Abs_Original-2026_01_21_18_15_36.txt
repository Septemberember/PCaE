public class Abs_Original {

    // Abs_Original: simple utility to compute absolute value of an int.
    // The implementation uses a straightforward conditional check.

    /*@ public normal_behavior
      @   requires num > Integer.MIN_VALUE;
      @   ensures (num < 0 ==> \result == -num) && (num >= 0 ==> \result == num);
      @   ensures \result >= 0;
      @ also
      @ public exceptional_behavior
      @   requires num == Integer.MIN_VALUE;
      @   signals (ArithmeticException) true;
      @*/
    public static int Abs(int num) {
        if (num < 0)
            return -num;
        else
            return num;
    }
}
