      @   // The logic ensures overhead is between 1 and 10 based on the divisions by 10
      @*/
    public static String calculateRatio(int fileSize) {
        String ratioStr = null;
        int level = 0;
        int overhead = 0;

        if (fileSize < 50) {
            fileSize = 50;
            overhead = overhead - 3;
            overhead *= fileSize / 10;
        } else {
            overhead = overhead + 1;
            overhead *= fileSize / 100;
        }

        if (fileSize > 400) {
            fileSize = 400;
            overhead = overhead + 5;
            overhead *= fileSize / 100;
        }

        if(overhead > 80){
            overhead = overhead > 100 ? 100 : overhead;
            overhead = overhead / 10;
        }else{
            overhead = overhead < 10 ? 10 : overhead;
            overhead = overhead / 10;
        }

        int ratio = fileSize / 4 + 30;

        if (ratio >= 45 && ratio <= 70) {
            level = (ratio - 45) / 5;
            if(level < 5) {
                ratioStr = new String("ratio:" + ratio);
            }
        } else {
            level = 5 + (ratio % 5);
            ratioStr = new String("ratio:" + ratio);
        }
        
        /*@  // To handle the case where ratioStr might be null if ratio is in [45, 70] but level >= 5
          @  // However, based on the math (70-45)/5 = 5, level < 5 is only false when ratio is exactly 70.
          @  // We ensure ratioStr is initialized if it was null.
          @*/
        if (ratioStr == null) {
            ratioStr = new String("ratio:" + ratio);
        }

        ratioStr.trim();
        ratioStr = ratioStr + " level:" + level + " overhead:" + overhead;
        return ratioStr;
    }
}
