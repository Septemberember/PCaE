public class SmallestEvenMul_Original {
    // This simple utility computes the smallest even multiple of n
    /* Helper note: result is n if even, otherwise 2*n. */

    /*@ public normal_behavior
      @   requires n > 0;
      @   ensures \result % 2 == 0;
      @   ensures \result % n == 0;
      @   ensures (n % 2 == 0) ==> (\result == n);
      @   ensures (n % 2 != 0) ==> (\result == 2 * n);
      @   ensures (\forall int i; n <= i && i < \result; i % 2 != 0 || i % n != 0);
      @   assignable \nothing;
      @*/
    public static int smallestEvenMultiple(int n) {
        return n % 2 == 0 ? n : 2 * n;
    }
}
