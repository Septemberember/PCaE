public class EchoIntLoop_Mutant5 {

    // Class EchoIntLoop_Mutant5 contains a simple integer echo operation
    // The echo method decrements a counter x times and returns the result
    /*@ public normal_behavior
      @   requires x >= 0;
      @   ensures \result == -x;
      @ also
      @ public normal_behavior
      @   requires x < 0;
      @   ensures \result == 0;
      @*/
    public static int echo(int x) {
        int res = 0;

        /*@ maintaining 0 <= i && i <= (x > 0 ? x : 0);
          @ maintaining res == -i;
          @ decreasing (x > 0 ? x - i : 0);
          @*/
        for(int i = 0; i < x; i++) {
            res = res - 1;
        }
        return res;
    }
}
