public class EchoIntLoop_Mutant3 {
    /* Simple, readable integer loop counting iterations up to x-1 */
    /*@ public normal_behavior
      @   ensures (x > 1) ==> (\result == x - 1);
      @   ensures (x <= 1) ==> (\result == 0);
      @   pure
      @*/
    public static int echo(int x) {
        int res = 0;
        // initialize the accumulator
        /*@ maintaining (x > 1 && i >= 1 && i <= x) ==> (res == i - 1);
          @ maintaining (x <= 1) ==> (res == 0);
          @ maintaining 1 <= i && (x > 1 ? i <= x : i == 1);
          @ decreases (x > i) ? (x - i) : 0;
          @*/
        for(int i = 1; i < x; i++) {
            // each iteration increments the accumulator by one
            res = res + 1;
        }
        // return the computed count (equals x-1 for x > 0)
        return res;
    }
}
