public class LeapYear_Mutant5 {

    /** Leap year calculation utility.
     *  This helper determines leap years using a 4-100-500 rule.
     *  See the inline implementation for exact conditional structure.
     */

    //@ ensures \result == true <==> (year % 4 == 0 && (year % 100 != 0 || year % 500 == 0));
    //@ ensures \result == false <==> (year % 4 != 0 || (year % 100 == 0 && year % 500 != 0));
    public static boolean isLeapYear(int year) {
        boolean leap = false;
        if (year % 4 == 0) {
            if (year % 100 == 0) {
                if (year % 500 == 0)
                    leap = true;
                else
                    leap = false;
            } else
                leap = true;
        } else
            leap = false;
        return leap;
    }
    // End of isLeapYear method

}
