public class Return100Nested_Mutant5 {

    // Utility: demonstrates a nested loop that increments 100 times
    /*@ public normal_behavior
      @ ensures \result == 100;
      @*/
    public static int return100 () {
        int res = 0;

        // Outer loop counts to ten
        /*@ maintaining 0 <= j && j <= 10;
          @ maintaining res == j * 10;
          @ decreasing 10 - j;
          @*/
        for(int j = 0; j < 10; j++) {

            // Inner loop counts to ten for each outer iteration
            /*@ maintaining 0 <= i && i <= 10;
              @ maintaining res == j * 10 + i;
              @ decreasing 10 - i;
              @*/
            for(int i = 0; i < 10; i++) {

                // Increment accumulator
                res = res + 1;
            }
        }

        // Final result is 100
        return res;
    }
}
