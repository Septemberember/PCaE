    public static int addLoop(int x, int y) {
        // Initialize the running total with the given y
        int sum = y;
        // Branch based on sign of x
        if (x > 0) {
            // Positive x: count up
            int n = 0;
            /*@ maintaining sum == y + (x - n);
              @ maintaining n >= 0;
              @ decreasing n;
              @*/
            for(n = x; n > 0; ) {
                sum = sum + 1;
                n = n - 1;
            }
        } else {
            // Negative or zero x: count down
            int n = 0;
            /*@ maintaining sum == y - ((-x) - n);
              @ maintaining n >= 0;
              @ decreasing n;
              @*/
            for(n = -x; n > 0; ) {
                sum = sum - 1;
                n = n - 1;
            }
        }
        return sum;
    }
