public class TemperatureAnalyzer {

    //@ spec_public
    public static double analyzeTemperature(int modifier) {
        //@ requires modifier != 2; // divisor would be 0 if modifier == 2
        //@ ensures \result == (temp > 50 ? 
        //@   (temp - 10) / (modifier - 1) + (temp - 10) * 0.1 + 1.3 + (temp - 10) - 3 + (average > 30 ? 2.0 : -1.0) :
        //@   temp * 0.2 + 1.3 + temp - 3 + (average < 10 ? 2.0 : -1.0));
        //@ ensures \result >= 0.0; // Based on the calculations, result should be non-negative
        int temp = 96;
        int offset = 5;
        double average = 0.0;
        boolean alert = false;

        temp = temp + modifier;
        temp = temp - offset;
        temp = temp * 2;

        int factor = 3;
        int divisor = modifier - 2;
        double result = 0.0;

        if (temp > 50) {
            divisor = divisor + 1;
            temp = temp - 10;
            result = (double) temp / divisor;
            average = result + temp * 0.1;
            alert = average > 30;
        } else {
            temp = temp + 15;
            divisor = divisor - 1;
            average = temp * 0.2;
            alert = average < 10;
        }

        double correction = 1.0;
        //@ maintaining 0 <= i && i <= 4;
        //@ maintaining correction == 1.0 + (\sum int j; 0 <= j && j < i; j * 0.05);
        //@ decreases 4 - i;
        for (int i = 0; i < 4; i++) {
            correction += i * 0.05;
        }
        //@ assert correction == 1.3; // 1.0 + 0*0.05 + 1*0.05 + 2*0.05 + 3*0.05 = 1.3

        double finalValue = average + correction + temp - factor;

        if (alert) {
            finalValue += 2.0;
        } else {
            finalValue -= 1.0;
        }
        return finalValue;
    }
}
