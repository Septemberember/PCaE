public class PowerOfTwoBranch_Original {

    //@ ensures \result == (n > 0 && (\exists int k; 0 <= k; n == (1 << k)));
    public static boolean isPowerOfTwo(int n) {
        if (n <= 0) {
            return false;
        }
        if ((n & (n - 1)) != 0) {
            return false;
        }
        return true;
    }
}
