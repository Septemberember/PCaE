        int overhead = 0;

        //@ maintaining fileSize >= 50;
        if (fileSize < 50) {
            fileSize = 50;
            overhead = overhead - 3;
            overhead *= fileSize / 10;
        } else {
            overhead = overhead + 1;
            overhead *= fileSize / 100;
        }

        //@ maintaining fileSize <= 400;
        if (fileSize > 400) {
            fileSize = 400;
            overhead = overhead + 5;
            overhead *= fileSize / 100;
        }

        //@ maintaining overhead >= 10;
        //@ maintaining overhead <= 10;
        if(overhead > 80){
            overhead = overhead > 100 ? 100 : overhead;
            overhead = overhead / 10;
        }else{
            overhead = overhead < 10 ? 10 : overhead;
            overhead = overhead / 10;
        }

        int ratio = fileSize / 4 + 30;

        //@ maintaining level >= 0;
        //@ maintaining level <= 9;
        if (ratio >= 45 && ratio <= 70) {
            level = (ratio - 45) / 5;
            if(level < 5) {
                ratioStr = new String("ratio:" + ratio);
            }
        } else {
            level = 5 + (ratio % 5);
            ratioStr = new String("ratio:" + ratio);
        }
        ratioStr.trim();
        ratioStr = ratioStr + " level:" + level + " overhead:" + overhead;
        return ratioStr;
    }
