    /*@ public normal_behavior
      @   requires modifier != 2; // divisor would be 0 if modifier == 2
      @   assignable \nothing;

      @                        (1.0 + 0.05 + 0.1 + 0.15) + (temp - 10) - 3 + 2.0) :
      @                       (temp * 0.2 + (1.0 + 0.05 + 0.1 + 0.15) + temp - 3 - 1.0))
      @           where {
      @               int temp = (96 + modifier - 5) * 2;
      @           };
      @*/
    public static double analyzeTemperature(int modifier) {
        int temp = 96;
        int offset = 5;
        double average = 0.0;

        temp = temp - offset;

            temp = temp - 10;

            average = result + temp * 0.1;
            alert = average > 30;
        } else {
            temp = temp + 15;
            divisor = divisor - 1;
            average = temp * 0.2;
            alert = average < 10;
        }

        double correction = 1.0;
        //@ maintaining 0 <= i && i <= 4;
        //@ maintaining correction == 1.0 + (\sum int j; 0 <= j && j < i; j * 0.05);
        //@ decreases 4 - i;
        for (int i = 0; i < 4; i++) {
            correction += i * 0.05;
        }

        double finalValue = average + correction + temp - factor;

        if (alert) {

        } else {
            finalValue -= 1.0;
        }
        return finalValue;
    }
