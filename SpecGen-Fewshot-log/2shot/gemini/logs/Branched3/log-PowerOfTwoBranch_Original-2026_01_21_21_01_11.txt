public class PowerOfTwoBranch_Original {

    //@ ensures \result <==> (n > 0 && (n & (n - 1)) == 0);
    //@ ensures \result <==> (n > 0 && (\exists int i; 0 <= i && i < 31; n == (1 << i)));
    public static boolean isPowerOfTwo(int n) {
        if (n <= 0) {
            return false;
        }
        if ((n & (n - 1)) != 0) {
            return false;
        }
        return true;
    }
}
