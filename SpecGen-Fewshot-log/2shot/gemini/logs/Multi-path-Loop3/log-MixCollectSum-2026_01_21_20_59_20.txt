import java.util.*;
public class MixCollectSum {
    //@ ensures (x > 220 || (x >= 195 && x != 200)) ==> \result >= 0;
    public static double mixSum(int x) {
        ArrayList<Integer> list = new ArrayList<>();
        int sum = 0;
        int times = 0;
        int tmp = 10;
        if(x > 220){
            x = 220;
        }
        //@ maintaining x < 195 || (x >= 49 && x <= 220);
        //@ maintaining list != null;
        //@ decreases x;
        while(x >= 195){
           if(x == 200){
               x = x - 150;
               sum = tmp / x;
           }
           x = x - 1;
           list.add(x);
        }
        //@ maintaining 0 <= i && i <= list.size();
        //@ decreases list.size() - i;
        for(int i : list){
            sum = sum + i;
        }
        return sum;
    }
}
